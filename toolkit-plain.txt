print("Welcome to My Mini-Toolkit!")

tasks=[]

def even_counter():
    nums=[1,2,3,4,5,6]
    c=0
    for n in nums:
        if n%2==0: c+=1
    print(f"{nums} -> {c} evens")

def doubler():
    for x in [1,2,3]:
        print(f"{x*2}")

def todo():
    while True:
        print(f"Tasks:{tasks} | 1.Add 2.View 3.Back")
        o=input("Choose: ")
        if o=="1": tasks.append(input("Task: "))
        elif o=="2": print(tasks)
        elif o=="3": break
        else: print(f"{o} invalid")

while True:
    print("\n=== MY MINI-TOOLKIT ===\n1.Even 2.Double 3.To-Do 4.Quit")
    ch=input("Choose 1-4: ")
    if ch=="1": even_counter()
    elif ch=="2": doubler()
    elif ch=="3": todo()
    elif ch=="4":
        print("Goodbye! Thanks!")
        break
    else: print(f"'{ch}' not on menu. Pick 1-4.")
